cmake_minimum_required(VERSION 3.21)
project(RelayChromium LANGUAGES CXX OBJCXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_OSX_DEPLOYMENT_TARGET 14.0)
find_package(Qt6 6.11 REQUIRED COMPONENTS WebEngineWidgets Widgets)
add_library(RelayChromium SHARED RelayChromium.mm)
target_link_libraries(RelayChromium PRIVATE Qt6::WebEngineWidgets Qt6::Widgets "-framework AppKit")
target_compile_options(RelayChromium PRIVATE -fobjc-arc -Wall -Wextra)
set_target_properties(RelayChromium PROPERTIES MACOSX_RPATH ON INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR ON)
